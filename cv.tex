\section*{Research Projects}

\textbf{Multi-Camera Array for Non-Lambertian Scene Capture}
\begin{itemize}
    \item Designed and built a synchronized 32-RGB-camera array for acquiring multi-view datasets of non-Lambertian scenes.
    \item Implemented hardware/software synchronization across all 32 cameras for consistent multi-view capture.
\end{itemize}

\textbf{Robust Neural Radiance Fields under Sparse Supervision}
\begin{itemize}
    \item Proposed a DIBR-based data augmentation method to improve NeRF performance in sparse-view scenarios.
    \item Introduced dense depth supervision to stabilize geometric learning.
    \item Developed a multi-task learning framework to mitigate noise introduced by augmented views.
\end{itemize}

\textbf{Diffusion-based Controllable Augmentation}
\begin{itemize}
    \item Implemented a controllable image augmentation framework using diffusion models for classification tasks.
\end{itemize}

\textbf{Medical Image Segmentation}
\begin{itemize}
    \item Evaluated semi-supervised and active learning strategies for histopathology segmentation with missing annotations.
\end{itemize}
